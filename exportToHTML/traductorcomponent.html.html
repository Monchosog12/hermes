traductorcomponent.html 
<main class="workbench">
  <h1>Traductor de código</h1>
  <p class="muted">Elige los lenguajes y el modelo de IA, escribe o carga tu código y traduce.</p>

  <form [formGroup]="form" (ngSubmit)="submit()" novalidate>
    <div class="selectors">
      <label class="field" for="sourceLanguage">Lenguaje de origen
        <select id="sourceLanguage" formControlName="sourceLanguage">
          @for (l of languages; track l.code) {
            <option [value]="l.code">{{ l.label }}</option>
          }
        </select>
      </label>

      <button class="btn btn-ghost btn-small swap" type="button" (click)="swap()"
              aria-label="Intercambiar lenguajes">⇄</button>

      <label class="field" for="targetLanguage">Lenguaje de destino
        <select id="targetLanguage" formControlName="targetLanguage">
          @for (l of languages; track l.code) {
            <option [value]="l.code">{{ l.label }}</option>
          }
        </select>
      </label>

      <label class="field" for="modelId">Modelo de IA
        <select id="modelId" formControlName="modelId">
          @for (m of ai.models(); track m.id) {
            <option [value]="m.id">{{ m.label }}</option>
          }
        </select>
      </label>
    </div>
    @if (ai.modelsError(); as message) {
      <p class="alert alert-error" role="alert">{{ message }}</p>
    }

    <div class="editor-slot">
      <app-codeeditorcomponent label="Código de origen" [(value)]="code" [language]="sourceLanguage()"
                               [allowUpload]="true" [maxChars]="maxChars"
                               (languageDetected)="onLanguageDetected($event)" />
    </div>

    @if (error(); as message) {
      <p class="alert alert-error" role="alert">{{ message }}</p>
    }
    <button class="btn submit" type="submit" [disabled]="loading()">
      {{ loading() ? 'Traduciendo…' : 'Traducir' }}
    </button>
  </form>

  @if (loading()) {
    <p class="muted" role="status">La IA está trabajando. Puede tardar hasta un minuto.</p>
  }

  @if (result(); as view) {
    <app-resultcomponent [view]="view" />
  }
</main>
